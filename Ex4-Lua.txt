function calculadora(a, b, op)
    if op == "+" then
        return a + b
    elseif op == "-" then
        return a - b
    elseif op == "*" then
        return a * b
    elseif op == "/" then
        if b ~= 0 then
            return a / b
        else
            return "Erro: divisão por zero"
        end
    else
        return "Erro: operador inválido"
    end
end

print(calculadora(10, 5, "+"))
print(calculadora(10, 5, "-"))
print(calculadora(10, 5, "*"))
print(calculadora(10, 5, "/"))
print(calculadora(10, 0, "/"))
print(calculadora(10, 5, "%"))
