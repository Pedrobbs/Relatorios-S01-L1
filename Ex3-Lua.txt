function filtrarPares(tabela)
    local pares = {}
    for i = 1, #tabela do
        if tabela[i] % 2 == 0 then
            table.insert(pares, tabela[i])
        end
    end
    return pares
end

local numeros = {5, 8, 13, 20, 42, 7, 16}
local resultado = filtrarPares(numeros)

print("Números pares encontrados:")
for i = 1, #resultado do
    io.write(resultado[i] .. " ")
end
